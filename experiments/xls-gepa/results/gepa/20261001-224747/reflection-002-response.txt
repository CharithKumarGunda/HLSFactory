// Implementation of 2D 8x8 inverse discrete cosine transform (IDCT) -- as a
// sequence of 8-point IDCTs -- for rows and columns of the 8x8 "Minimum Coded
// Unit" (MCU) coefficients.
//
// This is a port of the go IDCT algorithm, which is ultimately a fixed-point
// version of Chen et al (1980): http://www.researchgate.net/publication/224654667_IDCT_in_Non-Recursive_Form

// Mathematical constants
const PI: i32 = 40000;
const PI_OVER_8: i32 = PI / 8;
const PI_OVER_4: i32 = PI / 4;
const PI_OVER_2: i32 = PI / 2;
const PI_OVER_6: i32 = PI / 6;
const PI_OVER_3: i32 = PI / 3;
const TWO_PI: i32 = PI * 2;
const SQRT_2: i32 = 16384; // 2^15

// Sine and cosine tables for the IDCT
let sin_table: [i32; 16] = [
    -16384, -7158, -2726, -1312, -456, 456, 1312, 2726, 7158, 16384, 16384, 7158, 2726, 1312, 456,
];
let cos_table: [i32; 16] = [
    16384, 14438, 12045, 9634, 7223, 4813, 2408, 1204, 480, 1204, 2408, 4813, 7223, 9634, 12045, 14438,
];

// Helper functions for IDCT
fn sin(x: i32) -> i32 {
    sin_table[(x + PI) % PI] / SQRT_2
}

fn cos(x: i32) -> i32 {
    cos_table[(x + PI) % PI] / SQRT_2
}

fn idct_row(row: &[i32]) -> [i32; 8] {
    let mut result = [0; 8];
    for (i, &val) in row.iter().enumerate() {
        let mut sum = 0;
        for (j, &table_val) in sin_table.iter().enumerate() {
            sum += table_val * val * cos(i * 8 + j);
        }
        result[i] = sum;
    }
    result
}

fn idct_col(col: &[i32]) -> [i32; 8] {
    let mut result = [0; 8];
    for (i, &val) in col.iter().enumerate() {
        let mut sum = 0;
        for (j, &table_val) in sin_table.iter().enumerate() {
            sum += table_val * val * cos(i * 8 + j);
        }
        result[i] = sum;
    }
    result
}

fn idct_block(block: &[i32; 64]) -> [i32; 64] {
    let mut result = [0; 64];
    for (i, row) in block.iter().enumerate() {
        let row_result = idct_row(row);
        for (j, &val) in row_result.iter().enumerate() {
            result[i * 8 + j] = val;
        }
    }
    for (i, col) in block.iter().enumerate() {
        let col_result = idct_col(col);
        for (j, &val) in col_result.iter().enumerate() {
            result[i * 8 + j] = val;
        }
    }
    result
}

pub fn idct(f: [i32; 64]) -> [i32; 64] {
    idct_block(&f)
}
